Широкие возможности ESP32 в умном доме

Использование ESP32 в умном доме открывает широкие возможности благодаря встроенному Wi-Fi и Bluetooth, а также высокой производительности микроконтроллера. ESP32 позволяет создавать умные устройства, которые могут быть интегрированы в единую сеть для автоматизации различных процессов в доме. Рассмотрим несколько примеров проектов на базе ESP32 для умного дома.

Пример 1: Умное освещение

Компоненты

  • ESP32 — основной контроллер проекта
  • Релейный модуль — для управления светом
  • Датчик освещенности (фоторезистор) — для автоматического включения света
  • Кнопка — для ручного управления (опционально)
  • Соединительные провода — для подключения компонентов
  • Резисторы 10 кОм — для подтяжки сигналов

Схема подключения

  1. Релейный модуль:
    • VCC → 3.3V на ESP32
    • GND → GND на ESP32
    • IN → GPIO 23 на ESP32 (для управления реле)
  2. Фоторезистор:
    • Один вывод → 3.3V на ESP32
    • Другой вывод → аналоговый вход GPIO 34 (A0) и через резистор 10 кОм к GND
  3. Кнопка (опционально):
    • Один контакт → GND
    • Другой контакт → GPIO 22 на ESP32 и через резистор 10 кОм к 3.3V
💡Подсказка: GPIO 34, 35, 36, 39 на ESP32 — это только аналоговые входы (ADC), они не поддерживают цифровой вывод или подтяжку.

Программирование

Скопируйте приведённый ниже код в Arduino IDE. Замените your_SSID и your_PASSWORD на данные вашей сети Wi-Fi.

#include <WiFi.h>

#define RELAY_PIN 23
#define LDR_PIN 34 // Аналоговый вход
#define BUTTON_PIN 22

const char* ssid = "your_SSID";
const char* password = "your_PASSWORD";

WiFiServer server(80);

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);
  pinMode(BUTTON_PIN, INPUT_PULLUP);

  digitalWrite(RELAY_PIN, LOW); // Свет выключен по умолчанию

  // Подключение к Wi-Fi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Connecting to WiFi...");
  }
  Serial.println("Connected to WiFi");

  server.begin();
  Serial.println("Server started");

  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());
}

void loop() {
  WiFiClient client = server.available();
  if (client) {
    Serial.println("New Client.");
    String currentLine = "";
    while (client.connected()) {
      if (client.available()) {
        char c = client.read();
        Serial.write(c);
        if (c == '\n') {
          if (currentLine.length() == 0) {
            client.println("HTTP/1.1 200 OK");
            client.println("Content-type:text/html");
            client.println();

            client.print("Click <a href=\"/H\">here</a> to turn the light on.<br>");
            client.print("Click <a href=\"/L\">here</a> to turn the light off.<br>");

            client.println();
            break;
          } else {
            currentLine = "";
          }
        } else if (c != '\r') {
          currentLine += c;
        }

        if (currentLine.endsWith("GET /H")) {
          digitalWrite(RELAY_PIN, HIGH);
        }
        if (currentLine.endsWith("GET /L")) {
          digitalWrite(RELAY_PIN, LOW);
        }
      }
    }
    client.stop();
    Serial.println("Client Disconnected.");
  }

  // Чтение значения с фоторезистора
  int ldrValue = analogRead(LDR_PIN);
  Serial.print("LDR Value: ");
  Serial.println(ldrValue);

  // Управление светом на основе освещенности
  if (ldrValue < 500) {
    digitalWrite(RELAY_PIN, HIGH); // Включаем свет
  } else {
    digitalWrite(RELAY_PIN, LOW); // Выключаем свет
  }

  // Проверка состояния кнопки
  if (digitalRead(BUTTON_PIN) == LOW) {
    digitalWrite(RELAY_PIN, !digitalRead(RELAY_PIN)); // Переключаем состояние реле
    delay(500); // Защита от дребезга контактов
  }
}

Пример 2: Умный термостат

Компоненты

  • ESP32 — основной контроллер
  • Датчик температуры и влажности DHT22 — для измерения параметров среды
  • Релейный модуль — для управления отоплением/охлаждением
  • LCD-дисплей с I2C — для локального отображения данных (опционально)

Схема подключения

  1. Датчик DHT22:
    • VCC → 3.3V на ESP32
    • GND → GND на ESP32
    • Data → GPIO 4 на ESP32
  2. Релейный модуль:
    • VCC → 3.3V на ESP32
    • GND → GND на ESP32
    • IN → GPIO 5 на ESP32 (для управления реле)
  3. LCD-дисплей с I2C (опционально):
    • VCC → 3.3V, GND → GND
    • SDA → GPIO 21, SCL → GPIO 22
ℹ️Примечание: Для работы с DHT22 установите библиотеку DHT sensor library от Adafruit через менеджер библиотек Arduino IDE.

Программирование

#include <WiFi.h>
#include <DHT.h>

#define DHTPIN 4
#define RELAY_PIN 5
#define DHTTYPE DHT22

const char* ssid = "your_SSID";
const char* password = "your_PASSWORD";

DHT dht(DHTPIN, DHTTYPE);
WiFiServer server(80);

void setup() {
  Serial.begin(115200);
  pinMode(RELAY_PIN, OUTPUT);

  digitalWrite(RELAY_PIN, LOW); // Отопление/охлаждение выключено по умолчанию

  dht.begin();

  // Подключение к Wi-Fi
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(1000);
    Serial.println("Connecting to WiFi...");
  }
  Serial.println("Connected to WiFi");

  server.begin();
  Serial.println("Server started");

  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());
}

void loop() {
  WiFiClient client = server.available();
  if (client) {
    Serial.println("New Client.");
    String currentLine = "";
    while (client.connected()) {
      if (client.available()) {
        char c = client.read();
        Serial.write(c);
        if (c == '\n') {
          if (currentLine.length() == 0) {
            client.println("HTTP/1.1 200 OK");
            client.println("Content-type:text/html");
            client.println();

            client.print("Click <a href=\"/H\">here</a> to turn the heating/cooling on.<br>");
            client.print("Click <a href=\"/L\">here</a> to turn the heating/cooling off.<br>");

            client.println();
            break;
          } else {
            currentLine = "";
          }
        } else if (c != '\r') {
          currentLine += c;
        }

        if (currentLine.endsWith("GET /H")) {
          digitalWrite(RELAY_PIN, HIGH);
        }
        if (currentLine.endsWith("GET /L")) {
          digitalWrite(RELAY_PIN, LOW);
        }
      }
    }
    client.stop();
    Serial.println("Client Disconnected.");
  }

  // Чтение значения температуры и влажности
  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();

  // Проверка корректности данных
  if (isnan(temperature) || isnan(humidity)) {
    Serial.println("Failed to read from DHT sensor!");
    return;
  }

  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.print(" *C, Humidity: ");
  Serial.print(humidity);
  Serial.println(" %");

  // Пример логики управления отоплением/охлаждением
  if (temperature < 22.0) {
    digitalWrite(RELAY_PIN, HIGH); // Включаем отопление/охлаждение
  } else if (temperature > 25.0) {
    digitalWrite(RELAY_PIN, LOW); // Выключаем отопление/охлаждение
  }

  delay(2000); // Задержка между проверками
}

Дальнейшие улучшения

После базовой реализации проектов можно расширить их функционал следующими способами:

  • Интеграция с голосовыми помощниками: Использование платформ, таких как Google Home или Amazon Alexa, для управления устройствами с помощью голосовых команд
  • Использование MQTT: Для более гибкого и масштабируемого обмена данными между устройствами умного дома
  • Создание пользовательского веб-интерфейса: Для управления и мониторинга всех устройств умного дома с помощью одного веб-приложения
  • Интеграция с датчиками движения: Для автоматического управления освещением и системами безопасности
  • Энергоменеджмент: Мониторинг и оптимизация энергопотребления всех устройств в умном доме
🚀Идея: Объедините оба проекта в единую систему: умное освещение + термостат + уведомления в Telegram = полноценный узел умного дома на ESP32!
